-- Table: public.car

-- DROP TABLE IF EXISTS public.car;

CREATE TABLE IF NOT EXISTS public.car
(
    id_car integer NOT NULL GENERATED BY DEFAULT AS IDENTITY ( INCREMENT 1 START 1 MINVALUE 1 MAXVALUE 2147483647 CACHE 1 ),
    id_status integer NOT NULL,
    id_rate integer NOT NULL,
    stamp character varying(100) COLLATE pg_catalog."default" NOT NULL,
    model character varying(100) COLLATE pg_catalog."default" NOT NULL,
    year_of_release smallint NOT NULL,
    colour character varying(100) COLLATE pg_catalog."default" NOT NULL,
    mileage numeric NOT NULL,
    state_number character varying(50) COLLATE pg_catalog."default" NOT NULL,
    CONSTRAINT car_pkey PRIMARY KEY (id_car),
    CONSTRAINT id_rate FOREIGN KEY (id_rate)
        REFERENCES public.rate (id_rate) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
        NOT VALID,
    CONSTRAINT id_status FOREIGN KEY (id_status)
        REFERENCES public.status (id_status) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
        NOT VALID
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS public.car
    OWNER to postgres;
